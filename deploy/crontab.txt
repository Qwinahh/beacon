# Beacon X Bot — Crontab (all times UTC)
# Install: (crontab -l 2>/dev/null; cat /path/to/beacon/deploy/crontab.txt) | crontab -
# View:    crontab -l
# Logs:    tail -f /path/to/beacon/deploy/logs/<job>.log
#
# Replace /root/beacon with wherever you cloned the repo.

BEACON=/root/beacon

# Alpha fast-track — every 30 min (urgent signals only, exits fast if nothing urgent)
*/30 * * * * bash $BEACON/deploy/run.sh alpha

# Main posts — fires hourly 08-22 UTC; post.py publishes ONLY during the auto-tuned
# active hours in data/growth/posting_schedule.json (capped at 6/day by MAX_POSTS_PER_DAY
# + cooldown). The exact hours are retuned weekly by the `retime` job below.
0 8-22 * * * bash $BEACON/deploy/run.sh post

# Engagement — replies, outbound, follow — every hour at :50
#   (~10 min before each post window at :00 — warms up account activity so early
#    engagement velocity primes the algorithm to distribute the upcoming post)
50 * * * * bash $BEACON/deploy/run.sh engage

# Knowledge — learn runs twice daily, refresh after learn
0  6  * * * bash $BEACON/deploy/run.sh learn
0  17 * * * bash $BEACON/deploy/run.sh learn
30 6  * * * bash $BEACON/deploy/run.sh refresh

# Daily — track at 06:00, inspiration at 07:00, digest at 22:30
0  6  * * * bash $BEACON/deploy/run.sh track
0  7  * * * bash $BEACON/deploy/run.sh inspiration
30 22 * * * bash $BEACON/deploy/run.sh digest

# Weekly — growth analysis Sunday 09:00, suggestions report Monday 07:00
0 9 * * 0 bash $BEACON/deploy/run.sh growth
0 7 * * 1 bash $BEACON/deploy/run.sh suggest

# Auto-tune posting hours from trailing performance-by-hour — Sunday 05:00 UTC
0 5 * * 0 bash $BEACON/deploy/run.sh retime
